"""Сборки гостя, публичные ссылки и проверка совместимости."""
import secrets
import uuid
from typing import Any

from fastapi import APIRouter, Depends, Path, Response
from pydantic import BaseModel, Field, field_validator
from sqlalchemy import delete, func, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.api.deps import api_error, current_session
from app.api.presenters import is_estimate, present_part, present_totals
from app.api.sessions import MAX_BUILDS_PER_SESSION
from app.database import get_db
from app.models.build import Build, BuildItem
from app.models.catalog import Category, Manufacturer, Part
from app.models.session import AnonSession
from app.services.build_check import UnknownPartsError, check_items, normalize_items

router = APIRouter(tags=["builds"])
SLUG_PATTERN = r"^[A-Za-z0-9_-]{8,16}$"


class ItemIn(BaseModel):
    part_id: int = Field(gt=0, lt=2**63)
    quantity: int = Field(default=1, ge=1, le=16)


class ItemsIn(BaseModel):
    items: list[ItemIn] = Field(default_factory=list, max_length=64)

    def pairs(self) -> list[tuple[int, int]]:
        return [(i.part_id, i.quantity) for i in self.items]


class BuildIn(ItemsIn):
    name: str = Field(min_length=1, max_length=100)
    description: str | None = Field(default=None, max_length=2000)

    @field_validator("name")
    @classmethod
    def _name(cls, v: str) -> str:
        v = v.strip()
        if not v:
            raise ValueError("Название не может быть пустым")
        return v

    @field_validator("description")
    @classmethod
    def _description(cls, v: str | None) -> str | None:
        return (v or "").strip() or None


async def _check(db: AsyncSession, pairs: list[tuple[int, int]]):
    try:
        return await check_items(db, pairs)
    except UnknownPartsError as exc:
        raise api_error(422, "unknown_parts", str(exc), part_ids=list(exc.part_ids)) from exc
    except ValueError as exc:
        raise api_error(422, "invalid_items", str(exc)) from exc


async def _estimate(db: AsyncSession, part_ids: list[int]) -> bool:
    if not part_ids:
        return False
    rows = (await db.execute(select(Part.source, Part.price).where(Part.id.in_(part_ids)))).all()
    return any(is_estimate(s, p) for s, p in rows)


async def _check_payload(db: AsyncSession, pairs: list[tuple[int, int]]) -> tuple[Any, dict[str, Any]]:
    result = await _check(db, pairs)
    ids = list({pid for pid, _ in pairs})
    totals = present_totals(result.total_price, item_count=len(ids),
                            unpriced_count=len(result.unpriced_part_ids), estimate=await _estimate(db, ids))
    return result, {**result.to_dict(), **totals}


@router.post("/compatibility/check")
async def check_compatibility(data: ItemsIn, db: AsyncSession = Depends(get_db)):
    """Проверка без сохранения. Токен не нужен."""
    _, payload = await _check_payload(db, data.pairs())
    return payload


async def _build_payload(db: AsyncSession, build: Build, *, owner: bool) -> dict[str, Any]:
    rows = (await db.execute(
        select(BuildItem, Part, Category.slug, Manufacturer.name)
        .join(Part, BuildItem.part_id == Part.id)
        .join(Category, Part.category_id == Category.id)
        .outerjoin(Manufacturer, Part.manufacturer_id == Manufacturer.id)
        .where(BuildItem.build_id == build.id)
        .order_by(Category.sort_order, BuildItem.id)
    )).all()
    items, available, unavailable = [], [], []
    for item, part, slug, maker in rows:
        ok = part.is_active and part.deleted_at is None
        (available if ok else unavailable).append((item.part_id, item.quantity))
        items.append({
            "part": present_part(part, slug, maker),
            "quantity": item.quantity,
            "price_snapshot": None if item.price_snapshot is None else str(item.price_snapshot),
            "is_available": ok,
        })
    # Снятые с продажи детали в проверку не идут, но остаются в составе со статусом is_available=false
    _, check = await _check_payload(db, available)
    payload = {
        "id": str(build.id),
        "name": build.name,
        "description": build.description,
        "is_public": build.is_public,
        "share_slug": build.share_slug if build.is_public else None,
        "share_path": f"/b/{build.share_slug}" if build.is_public else None,
        "created_at": build.created_at.isoformat(),
        "updated_at": build.updated_at.isoformat(),
        "items": items,
        "unavailable_part_ids": [pid for pid, _ in unavailable],
        "check": check,
    }
    payload.update({k: check[k] for k in ("total_price_display", "total_price_is_estimate",
                                          "total_price_note", "unpriced_count")})
    payload["total_price"] = check["total_price"]
    payload["is_owner"] = owner
    return payload


async def _save(db: AsyncSession, build: Build, data: BuildIn) -> None:
    pairs = data.pairs()
    result = await _check(db, pairs)
    if result.errors:
        raise api_error(422, "incompatible_build", "Сборка несовместима, исправьте ошибки перед сохранением",
                        check=result.to_dict())
    items = normalize_items(pairs)
    prices = dict((await db.execute(select(Part.id, Part.price).where(Part.id.in_(list(items))))).all()) \
        if items else {}
    build.name = data.name
    build.description = data.description
    all_unpriced = bool(items) and len(result.unpriced_part_ids) >= len(items)
    build.total_price = None if all_unpriced else result.total_price
    build.total_power_w = result.power.peak_w
    build.updated_at = func.now()
    await db.flush()
    await db.execute(delete(BuildItem).where(BuildItem.build_id == build.id))
    db.add_all(BuildItem(build_id=build.id, part_id=pid, quantity=qty, price_snapshot=prices.get(pid))
               for pid, qty in items.items())
    await db.commit()
    await db.refresh(build)


async def _owned(db: AsyncSession, s: AnonSession, build_id: uuid.UUID) -> Build:
    build = await db.scalar(select(Build).where(Build.id == build_id, Build.session_id == s.id))
    if build is None:
        raise api_error(404, "build_not_found", "Сборка не найдена")
    return build

@router.post("/builds", status_code=201)
async def create_build(data: BuildIn, s: AnonSession = Depends(current_session),
                       db: AsyncSession = Depends(get_db)):
    # Блокировка строки сессии: два параллельных запроса не обойдут лимит
    await db.execute(select(AnonSession.id).where(AnonSession.id == s.id).with_for_update())
    count = await db.scalar(select(func.count()).select_from(Build).where(Build.session_id == s.id))
    if count >= MAX_BUILDS_PER_SESSION:
        raise api_error(409, "build_limit", f"Можно сохранить не больше {MAX_BUILDS_PER_SESSION} сборок",
                        limit=MAX_BUILDS_PER_SESSION)
    build = Build(session_id=s.id, name=data.name, description=data.description)
    db.add(build)
    await _save(db, build, data)
    return await _build_payload(db, build, owner=True)


@router.get("/builds/{build_id}")
async def get_build(build_id: uuid.UUID, s: AnonSession = Depends(current_session),
                    db: AsyncSession = Depends(get_db)):
    return await _build_payload(db, await _owned(db, s, build_id), owner=True)


@router.put("/builds/{build_id}")
async def update_build(build_id: uuid.UUID, data: BuildIn, s: AnonSession = Depends(current_session),
                       db: AsyncSession = Depends(get_db)):
    build = await _owned(db, s, build_id)
    await _save(db, build, data)
    return await _build_payload(db, build, owner=True)


@router.delete("/builds/{build_id}", status_code=204)
async def delete_build(build_id: uuid.UUID, s: AnonSession = Depends(current_session),
                       db: AsyncSession = Depends(get_db)):
    await db.delete(await _owned(db, s, build_id))
    await db.commit()
    return Response(status_code=204)


@router.post("/builds/{build_id}/share")
async def share_build(build_id: uuid.UUID, s: AnonSession = Depends(current_session),
                      db: AsyncSession = Depends(get_db)):
    build = await _owned(db, s, build_id)
    if not build.share_slug:
        build.share_slug = secrets.token_urlsafe(9)  # 12 символов, ~72 бита
    build.is_public = True
    await db.commit()
    return {"share_slug": build.share_slug, "share_path": f"/b/{build.share_slug}",
            "api_path": f"/api/v1/shared/{build.share_slug}"}


@router.delete("/builds/{build_id}/share", status_code=204)
async def unshare_build(build_id: uuid.UUID, s: AnonSession = Depends(current_session),
                        db: AsyncSession = Depends(get_db)):
    build = await _owned(db, s, build_id)
    # Старая ссылка перестаёт работать; при повторной публикации будет новая
    build.is_public = False
    build.share_slug = None
    await db.commit()
    return Response(status_code=204)


@router.get("/shared/{slug}")
async def get_shared(slug: str = Path(pattern=SLUG_PATTERN), db: AsyncSession = Depends(get_db)):
    build = await db.scalar(select(Build).where(Build.share_slug == slug, Build.is_public.is_(True)))
    if build is None:
        raise api_error(404, "build_not_found", "Сборка не найдена или ссылка отозвана")
    return await _build_payload(db, build, owner=False)

"""Каталог: категории, производители, поиск и фильтры деталей."""
from decimal import Decimal
from typing import Literal

from fastapi import APIRouter, Depends, Path, Query
from sqlalchemy import and_, func, or_, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.api.deps import api_error
from app.api.presenters import present_part
from app.database import get_db
from app.models.catalog import Category, Manufacturer, Part

router = APIRouter(tags=["catalog"])


def _available():
    return and_(Part.is_active.is_(True), Part.deleted_at.is_(None))


def _escape_like(value: str) -> str:
    return value.replace("\\", "\\\\").replace("%", "\\%").replace("_", "\\_")


def _base_query():
    return (
        select(Part, Category.slug, Manufacturer.name)
        .join(Category, Part.category_id == Category.id)
        .outerjoin(Manufacturer, Part.manufacturer_id == Manufacturer.id)
        .where(_available())
    )


@router.get("/categories")
async def list_categories(db: AsyncSession = Depends(get_db)):
    rows = (await db.execute(
        select(Category, func.count(Part.id))
        .outerjoin(Part, and_(Part.category_id == Category.id, _available()))
        .group_by(Category.id)
        .order_by(Category.sort_order, Category.id)
    )).all()
    return [
        {"slug": c.slug, "name": c.name, "is_required": c.is_required, "max_items": c.max_items, "part_count": n}
        for c, n in rows
    ]


@router.get("/manufacturers")
async def list_manufacturers(category: str | None = Query(None, max_length=32), db: AsyncSession = Depends(get_db)):
    stmt = (select(Manufacturer.slug, Manufacturer.name, func.count(Part.id))
            .join(Part, Part.manufacturer_id == Manufacturer.id)
            .join(Category, Part.category_id == Category.id)
            .where(_available()))
    if category:
        stmt = stmt.where(Category.slug == category)
    rows = (await db.execute(stmt.group_by(Manufacturer.id).order_by(Manufacturer.name))).all()
    return [{"slug": s, "name": n, "part_count": c} for s, n, c in rows]


@router.get("/parts")
async def list_parts(
    category: str | None = Query(None, max_length=32),
    q: str | None = Query(None, max_length=100, description="поиск по названию"),
    manufacturer: str | None = Query(None, max_length=64, description="slug производителя"),
    min_price: Decimal | None = Query(None, ge=0),
    max_price: Decimal | None = Query(None, ge=0),
    has_price: bool | None = Query(None, description="false — только позиции без цены"),
    socket: str | None = Query(None, max_length=32),
    form_factor: str | None = Query(None, max_length=32),
    memory_type: str | None = Query(None, max_length=16),
    sort: Literal["name", "price", "-price"] = "name",
    limit: int = Query(24, ge=1, le=100),
    offset: int = Query(0, ge=0, le=100_000),
    db: AsyncSession = Depends(get_db),
):
    conds = []
    if category:
        conds.append(Category.slug == category)
    if q and q.strip():
        conds.append(Part.name.ilike(f"%{_escape_like(q.strip())}%", escape="\\"))
    if manufacturer:
        conds.append(Manufacturer.slug == manufacturer)
    if min_price is not None:
        conds.append(Part.price >= min_price)
    if max_price is not None:
        conds.append(Part.price <= max_price)
    if has_price is not None:
        conds.append(Part.price.is_not(None) if has_price else Part.price.is_(None))
    # Фильтры по характеристикам: одиночное значение или вхождение в список (кулеры, корпуса, CPU)
    if socket:
        conds.append(or_(Part.specs.contains({"socket": socket}), Part.specs.contains({"sockets": [socket]})))
    if form_factor:
        conds.append(or_(Part.specs.contains({"form_factor": form_factor}),
                         Part.specs.contains({"motherboard_form_factors": [form_factor]})))
    if memory_type:
        conds.append(or_(Part.specs.contains({"memory_type": memory_type}),
                         Part.specs.contains({"memory_types": [memory_type]})))
    stmt = _base_query().where(*conds)
    total = await db.scalar(select(func.count()).select_from(stmt.subquery()))
    order = {
        "name": (Part.name.asc(), Part.id.asc()),
        "price": (Part.price.asc().nulls_last(), Part.id.asc()),
        "-price": (Part.price.desc().nulls_last(), Part.id.asc()),
    }[sort]
    rows = (await db.execute(stmt.order_by(*order).limit(limit).offset(offset))).all()
    return {"items": [present_part(p, slug, m) for p, slug, m in rows], "total": total,
            "limit": limit, "offset": offset}


@router.get("/parts/{part_id}")
async def get_part(part_id: int = Path(gt=0, lt=2**63), db: AsyncSession = Depends(get_db)):
    row = (await db.execute(_base_query().where(Part.id == part_id))).first()
    if row is None:
        raise api_error(404, "part_not_found", "Деталь не найдена")
    return present_part(*row)

===== backend\app\api\deps.py =====
"""Общие зависимости API: анонимная сессия по Bearer-токену и единый формат ошибок."""
import hashlib
import secrets
from datetime import datetime, timedelta, timezone
from typing import Any

from fastapi import Depends, Header, HTTPException
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.database import get_db
from app.models.session import AnonSession

SESSION_TTL = timedelta(days=180)
TOUCH_INTERVAL = timedelta(hours=1)


def api_error(status: int, code: str, message: str, **extra: Any) -> HTTPException:
    return HTTPException(status_code=status, detail={"code": code, "message": message, **extra})


def utcnow() -> datetime:
    return datetime.now(timezone.utc)


def _aware(dt: datetime) -> datetime:
    return dt if dt.tzinfo else dt.replace(tzinfo=timezone.utc)


def hash_token(token: str) -> str:
    return hashlib.sha256(token.encode()).hexdigest()


def new_token() -> str:
    # 32 байта энтропии; в БД попадает только SHA-256
    return secrets.token_urlsafe(32)


async def current_session(
    authorization: str | None = Header(default=None),
    db: AsyncSession = Depends(get_db),
) -> AnonSession:
    if not authorization or not authorization[:7].lower() == "bearer ":
        raise api_error(401, "no_token", "Нужен anon_token: получите его через POST /api/v1/session")
    token = authorization[7:].strip()
    if not token or len(token) > 128:
        raise api_error(401, "invalid_token", "Токен недействителен")
    session = await db.scalar(select(AnonSession).where(AnonSession.token_hash == hash_token(token)))
    now = utcnow()
    if session is None or _aware(session.expires_at) <= now:
        raise api_error(401, "invalid_token", "Токен недействителен или истёк")
    # Продлеваем сессию не чаще раза в час, чтобы не писать в БД на каждый запрос
    if now - _aware(session.last_seen_at) > TOUCH_INTERVAL:
        session.last_seen_at = now
        session.expires_at = now + SESSION_TTL
        await db.commit()
    return session

===== backend\app\api\presenters.py =====
"""Представление деталей и сумм для API: рубли, «ориентировочная цена», «нет данных»."""
from decimal import Decimal
from typing import Any

from app.catalog_specs import POWER_REQUIRED_CATEGORIES, REQUIRED_SPEC_KEYS, SPEC_LABELS
from app.importers.docyx import SOURCE as DOCYX_SOURCE

NO_DATA = "нет данных"
ESTIMATE_NOTE = "ориентировочная цена"
NBSP = "\u00a0"
_SIGN = {"RUB": "₽", "USD": "$", "EUR": "€"}
_UNITS = (("_mm", "мм"), ("_w", "Вт"), ("_gb", "ГБ"), ("_mhz", "МГц"))
_ORDER = {k: i for i, k in enumerate(SPEC_LABELS)}


def is_empty(value: Any) -> bool:
    return value is None or value == "" or value == [] or value == {}


def format_money(value: Decimal | None, currency: str = "RUB") -> str:
    if value is None:
        return NO_DATA
    v = Decimal(value)
    text = f"{v:,.0f}" if v == v.to_integral_value() else f"{v:,.2f}"
    text = text.replace(",", NBSP).replace(".", ",")
    return f"{text}{NBSP}{_SIGN.get(currency, currency)}"


def format_spec(key: str, value: Any) -> str:
    if is_empty(value):
        return NO_DATA
    if isinstance(value, bool):
        return "есть" if value else "нет"
    if isinstance(value, (list, tuple)):
        return ", ".join(str(v) for v in value)
    if isinstance(value, (int, float, Decimal)):
        for suffix, unit in _UNITS:
            if key.endswith(suffix):
                return f"{value}{NBSP}{unit}"
        return str(value)
    return str(value)


def is_estimate(source: str | None, price: Decimal | None) -> bool:
    """Цены из датасета пересчитаны из USD, поэтому они ориентировочные."""
    return price is not None and source == DOCYX_SOURCE


def present_part(part: Any, category: str, manufacturer: str | None) -> dict[str, Any]:
    specs = dict(part.specs or {})
    required = REQUIRED_SPEC_KEYS.get(category, frozenset())
    keys = sorted(required | set(specs), key=lambda k: (_ORDER.get(k, len(_ORDER)), k))
    missing = [k for k in keys if k in required and is_empty(specs.get(k))]
    if category in POWER_REQUIRED_CATEGORIES and part.power_w is None:
        missing.append("power_w")
    estimate = is_estimate(part.source, part.price)
    return {
        "id": part.id,
        "category": category,
        "name": part.name,
        "manufacturer": manufacturer,
        "manufacturer_display": manufacturer or NO_DATA,
        "source": part.source,
        "price": None if part.price is None else str(part.price),
        "currency": part.currency,
        "has_price": part.price is not None,
        "price_display": format_money(part.price, part.currency),
        "price_is_estimate": estimate,
        "price_note": ESTIMATE_NOTE if estimate else None,
        "price_updated_at": part.price_updated_at.isoformat() if part.price_updated_at else None,
        "power_w": part.power_w,
        "power_display": NO_DATA if part.power_w is None else f"{part.power_w}{NBSP}Вт",
        "specs": specs,
        "specs_display": [
            {"key": k, "label": SPEC_LABELS.get(k, k), "value": format_spec(k, specs.get(k)),
             "missing": is_empty(specs.get(k))}
            for k in keys
        ],
        "missing_specs": missing,
        "is_data_complete": not missing and part.price is not None,
    }


def present_totals(total: Decimal | None, *, item_count: int, unpriced_count: int, estimate: bool,
                   currency: str = "RUB") -> dict[str, Any]:
    if item_count == 0:
        display = format_money(Decimal(0), currency)
    elif unpriced_count >= item_count:
        display = NO_DATA
    else:
        display = format_money(total, currency)
    notes = []
    if estimate:
        notes.append(ESTIMATE_NOTE)
    if 0 < unpriced_count < item_count:
        notes.append(f"без учёта позиций без цены: {unpriced_count}")
    return {
        "total_price_display": display,
        "total_price_is_estimate": estimate,
        "total_price_note": "; ".join(notes) or None,
        "unpriced_count": unpriced_count,
    }

===== backend\app\api\sessions.py =====
"""Выдача анонимного токена гостя."""
import logging

from fastapi import APIRouter, Depends, Request
from sqlalchemy import func, select
from sqlalchemy.ext.asyncio import AsyncSession

from app.api.deps import SESSION_TTL, api_error, current_session, hash_token, new_token, utcnow
from app.database import get_db
from app.models.build import Build
from app.models.session import AnonSession
from app.redis_client import redis_client

logger = logging.getLogger(__name__)
router = APIRouter(tags=["session"])

SESSIONS_PER_HOUR_PER_IP = 30
MAX_BUILDS_PER_SESSION = 15


async def session_rate_limit(request: Request) -> None:
    """Не больше 30 новых сессий в час с одного IP. Если Redis недоступен, лимит не применяется."""
    ip = request.client.host if request.client else "unknown"
    key = f"rl:session:{ip}"
    try:
        async with redisclient.pipeline(transaction=True) as pipe:
            pipe.set(key, 0, ex=3600, nx=True)
            pipe.incr(key)
            _, count = await pipe.execute()
    except Exception:  # noqa: BLE001
        logger.warning("Rate limit skipped: Redis unavailable", exc_info=True)
        return
    if count > SESSIONS_PER_HOUR_PER_IP:
        raise api_error(429, "too_many_sessions", "Слишком много новых сессий, попробуйте позже")


@router.post("/session", status_code=201)
async def create_session(_: None = Depends(session_rate_limit), db: AsyncSession = Depends(get_db)):
    token = new_token()
    expires = utcnow() + SESSION_TTL
    db.add(AnonSession(token_hash=hash_token(token), expires_at=expires))
    await db.commit()

            "builds_limit": MAX_BUILDS_PER_SESSION}

@dataclass(frozen=True, slots=True)
class MappedPart:
    category: str
    external_id: str
    name: str
    manufacturer: tuple[str, str] | None  # (slug, name)
    specs: dict[str, Any]
    power_w: int | None

# ---------- разбор значений ----------

def _num(v: Any) -> float | None:
    if isinstance(v, bool) or not isinstance(v, int | float) or not math.isfinite(v) or v < 0:
        return None
    return v


def _int(v: Any) -> int | None:
    n = _num(v)
    return None if n is None else int(round(n))


def _max_num(v: Any) -> float | None:
    """rpm и noise бывают числом или диапазоном [min, max]."""
    if isinstance(v, list):
        nums = [n for n in map(_num, v) if n is not None]
        return max(nums) if nums else None
    return _num(v)


def _pair(v: Any) -> tuple[float, float] | None:
    if isinstance(v, list) and len(v) == 2 and all(_num(x) is not None for x in v):
        return v[0], v[1]
    return None


def _str(v: Any) -> str | None:
    return v.strip()[:64] if isinstance(v, str) and v.strip() else None


def _clean(specs: dict[str, Any]) -> dict[str, Any]:
    return {k: v for k, v in specs.items() if v is not None}


def _price(v: Any) -> Decimal | None:
    n = _num(v)
    if not n:
        return None
    return Decimal(str(n)).quantize(Decimal("0.01"))


def gpu_tdp(chipset: str) -> int | None:
    for pattern, watts in _GPU_PATTERNS:
        if pattern.search(chipset):
            return watts
    return None


def manufacturer_of(name: str) -> tuple[str, str] | None:
    low = name.lower()
    raw = next((name[: len(b)] for b in TWO_WORD_BRANDS if low == b or low.startswith(b + " ")), None)
    if raw is None:
        raw = name.split()[0] if name.split() else ""
    slug = SLUG_ALIASES.get(raw.lower()) or re.sub(r"[^a-z0-9]+", "-", raw.lower()).strip("-")
    return (slug[:64], raw[:64]) if slug else None


def external_id(file: str, rec: dict[str, Any]) -> str:
    """Стабильный ID: хеш записи без цены, чтобы смена цены не плодила дубли."""
    core = {k: v for k, v in rec.items() if k not in ("price", "price_per_gb")}
    digest = hashlib.sha1(json.dumps(core, sort_keys=True, ensure_ascii=False).encode()).hexdigest()[:20]
    return f"{file}:{digest}"


def _capacity_label(gb: int) -> str:
    return f"{gb // 1000} ТБ" if gb >= 1000 and gb % 1000 == 0 else f"{gb} ГБ"


# ---------- маппинг категорий: (specs, power_w, дополнение к названию) ----------

Mapped = tuple[dict[str, Any], int | None, str | None]


def map_cpu(rec: dict[str, Any]) -> Mapped:
    if _CPU_EXCLUDE.search(rec["name"]):
        raise Skip("unsupported_platform")
    socket = ARCH_SOCKET.get(rec.get("microarchitecture") or "")
    if socket is None:
        raise Skip("unsupported_platform")
    tdp = _int(rec.get("tdp"))
    if not tdp:
        raise Skip("missing_tdp")
    power = max(tdp, min(round(tdp * CPU_POWER_FACTOR[socket]), CPU_POWER_CAP_W))
    specs = _clean({
        "socket": socket, "memory_types": list(SOCKET_MEMORY[socket]), "tdp_w": tdp,
        "integrated_graphics": bool(rec.get("graphics")), "graphics": _str(rec.get("graphics")),
        "cores": _int(rec.get("core_count")), "base_clock_ghz": _num(rec.get("core_clock")),
        "boost_clock_ghz": _num(rec.get("boost_clock")), "microarchitecture": rec.get("microarchitecture"),
        "power_estimated": True,
    })
    return specs, power, None


def map_motherboard(rec: dict[str, Any]) -> Mapped:
    socket = rec.get("socket")
    if socket not in SUPPORTED_SOCKETS:
        raise Skip("unsupported_platform")
    ff = MB_FORM_FACTORS.get(rec.get("form_factor") or "")
    if ff is None:
        raise Skip("unsupported_form_factor")
    memory = SOCKET_MEMORY[socket]
    # LGA1700 бывает и DDR4, и DDR5: DDR4-версии помечены в названии (DDR4 / D4)
    mtype = ("DDR4" if _DDR4_HINT.search(rec["name"]) else "DDR5") if len(memory) > 1 else memory[0]
    specs = _clean({
        "socket": socket, "form_factor": ff, "memory_type": mtype,
        "memory_slots": _int(rec.get("memory_slots")), "max_memory_gb": _int(rec.get("max_memory")),
        "color": _str(rec.get("color")),
    })
    return specs, None, None


def map_ram(rec: dict[str, Any]) -> Mapped:
    speed, modules = _pair(rec.get("speed")), _pair(rec.get("modules"))
    if not speed or int(speed[0]) not in (4, 5):
        raise Skip("unsupported_memory_type")
    if not modules or int(modules[0]) < 1 or int(modules[1]) < 1:
        raise Skip("missing_modules")
    gen, mhz, n, size = int(speed[0]), int(speed[1]), int(modules[0]), int(modules[1])
    specs = _clean({
        "memory_type": f"DDR{gen}", "speed_mhz": mhz, "modules": n, "module_gb": size,
        "capacity_gb": n * size, "cas_latency": _num(rec.get("cas_latency")), "color": _str(rec.get("color")),
    })
    return specs, 4 * n, f"DDR{gen}-{mhz} ({n}x{size} ГБ)"


def map_gpu(rec: dict[str, Any]) -> Mapped:
    chipset = rec.get("chipset") or ""
    tdp = gpu_tdp(chipset)
    if tdp is None:
        raise Skip("unknown_gpu_chipset")
    specs = _clean({
        "chipset": chipset, "memory_gb": _int(rec.get("memory")), "length_mm": _int(rec.get("length")),
        "core_clock_mhz": _int(rec.get("core_clock")), "boost_clock_mhz": _int(rec.get("boost_clock")),
        "tdp_w": tdp, "recommended_psu_w": math.ceil((tdp * 1.6 + 300) / 50) * 50,
        "color": _str(rec.get("color")), "power_estimated": True,
    })
    extra = None if chipset.lower() in rec["name"].lower() else chipset
    return specs, tdp, extra


def map_storage(rec: dict[str, Any]) -> Mapped:
    bus = rec.get("interface") or ""
    if bus.startswith("M.2 PCIe"):
        interface = "nvme"
    elif bus.startswith("SATA"):
        interface = "sata"
    else:
        raise Skip("unsupported_interface")  # M.2 SATA, U.2, PCIe-карты
    capacity = _int(rec.get("capacity"))
    if not capacity:
        raise Skip("missing_capacity")
    kind = "ssd" if rec.get("type") == "SSD" else "hdd"
    specs = _clean({
        "interface": interface, "bus": bus, "capacity_gb": capacity, "kind": kind,
        "rpm": _int(rec.get("type")) if kind == "hdd" else None,
        "form_factor": _str(str(rec["form_factor"])) if rec.get("form_factor") is not None else None,
        "cache_mb": _int(rec.get("cache")),
    })
    return specs, 6 if kind == "ssd" else 9, _capacity_label(capacity)


def map_psu(rec: dict[str, Any]) -> Mapped:
    ff = PSU_FORM_FACTORS.get(rec.get("type") or "")
    if ff is None:
        raise Skip("unsupported_form_factor")
    watts = _int(rec.get("wattage"))
    if not watts:
        raise Skip("missing_wattage")
    specs = _clean({
        "wattage_w": watts, "form_factor": ff, "efficiency": _str(rec.get("efficiency")),
        "modular": _str(rec.get("modular")), "color": _str(rec.get("color")),
    })
    return specs, None, None


def map_case(rec: dict[str, Any]) -> Mapped:
    t = rec.get("type") or ""
    if t.startswith("ATX"):
        boards = ["ATX", "Micro-ATX", "Mini-ITX"] + (["E-ATX"] if "Full Tower" in t else [])
    elif t.startswith("MicroATX"):
        boards = ["Micro-ATX", "Mini-ITX"]
    elif t.startswith("Mini ITX"):
        boards = ["Mini-ITX"]
    else:
        raise Skip("unsupported_case_type")
    # У компактных корпусов формат БП в датасете не указан: не угадываем
    psu = ["ATX", "SFX", "SFX-L"] if "Tower" in t and not t.startswith("Mini ITX") else None
    specs = _clean({
        "motherboard_form_factors": boards, "psu_form_factors": psu, "case_type": t,
        "side_panel": _str(rec.get("side_panel")), "volume_l": _num(rec.get("external_volume")),
        "bays_35": _int(rec.get("internal_35_bays")), "included_psu_w": _int(rec.get("psu")),
        "color": _str(rec.get("color")),
    })
    return specs, None, None


def map_cooler(rec: dict[str, Any]) -> Mapped:
    radiator = _int(rec.get("size"))
    specs = _clean({
        "type": "aio" if radiator else "air", "radiator_mm": radiator,
        "max_rpm": _int(_max_num(rec.get("rpm"))), "noise_db": _max_num(rec.get("noise_level")),
        "color": _str(rec.get("color")),
    })
    return specs, 15 if radiator else 4, None


MAPPERS: dict[str, Callable[[dict[str, Any]], Mapped]] = {
    "cpu": map_cpu, "motherboard": map_motherboard, "ram": map_ram, "gpu": map_gpu,
    "storage": map_storage, "psu": map_psu, "case": map_case, "cooler": map_cooler,
}


def map_records(category: str, records: Iterable[Any], *, include_unpriced: bool = False
                ) -> tuple[list[MappedPart], Counter]:
    file, mapper = FILES[category], MAPPERS[category]
    out: dict[str, MappedPart] = {}
    skipped: Counter = Counter()
    for rec in records:
        if not isinstance(rec, dict) or not isinstance(rec.get("name"), str) or not rec["name"].strip():
            skipped["invalid_record"] += 1
            continue
        price = _price(rec.get("price"))
        if price is None and not include_unpriced:
            skipped["no_price"] += 1
            continue
        try:
            specs, power, extra = mapper(rec)
        except Skip as exc:
            skipped[exc.reason] += 1
            continue
        base = " ".join(rec["name"].split())
        name = f"{base} {extra}" if extra else base
        eid = external_id(file, rec)
        prev = out.get(eid)
        if prev is not None:
            skipped["duplicate"] += 1
            if price is not None and (prev.price_usd is None or price < prev.price_usd):
                out[eid] = replace(prev, price_usd=price)
            continue
        out[eid] = MappedPart(category, eid, name[:255], manufacturer_of(base), specs, power, price)
    return list(out.values()), skipped


# ---------- загрузка ----------

def load_dataset(path: Path | None = None, *, ref: str = "main", categories: Iterable[str] | None = None,
                 timeout: float = 60) -> dict[str, list[Any]]:
    """Из локальной копии репозитория (path) или с raw.githubusercontent.com (ref: ветка или коммит)."""
    cats = list(categories or FILES)
    unknown = [c for c in cats if c not in FILES]
    if unknown:
        raise ValueError(f"Неизвестные категории: {', '.join(unknown)}")
    if path is not None and (path / "data" / "json").is_dir():
        path = path / "data" / "json"
    if path is None and not _REF_RE.fullmatch(ref):
        raise ValueError("Некорректный ref")
    data: dict[str, list[Any]] = {}
    for cat in cats:
        file = FILES[cat]
        if path is not None:
            raw = json.loads((path / f"{file}.json").read_text(encoding="utf-8"))
        else:
            url = RAW_URL.format(ref=urllib.parse.quote(ref, safe="/"), file=file)
            with urllib.request.urlopen(url, timeout=timeout) as resp:  # noqa: S310 (фиксированный https-хост)
                raw = json.load(resp)
        if not isinstance(raw, list):
            raise ValueError(f"{file}.json: ожидался JSON-массив")
        data[cat] = raw
    return data


def _chunks(items: list[Any], size: int) -> Iterator[list[Any]]:
    for i in range(0, len(items), size):
        yield items[i:i + size]


# ---------- запись в БД ----------

async def import_catalog(session, data: dict[str, list[Any]], *, usd_rate: Decimal | None = None,
                         include_unpriced: bool = False, retire_missing: bool = True,
                         retire_seed: bool = False) -> dict[str, Any]:
    """Upsert по (source, external_id), история цен, деактивация исчезнувших. Commit делает вызывающий."""
    from sqlalchemy import case, func, insert, select, text, true, update
    from sqlalchemy.dialects.postgresql import insert as pg_insert

    from app.models import Category, Manufacturer, Part, PartPriceHistory

    if not await session.scalar(text("SELECT pg_try_advisory_xact_lock(:k)"), {"k": LOCK_KEY}):
        raise ImportBusyError("Импорт каталога уже выполняется")
    if usd_rate is not None and usd_rate <= 0:
        raise ValueError("Курс должен быть больше нуля")

    currency = "RUB" if usd_rate else "USD"
    cat_ids = dict((await session.execute(select(Category.slug, Category.id))).all())
    stats: dict[str, Any] = {"source": SOURCE, "currency": currency,
                             "usd_rate": str(usd_rate) if usd_rate else None, "categories": {},
                             "created": 0, "updated": 0, "price_changes": 0, "retired": 0, "seed_retired": 0}
    mapped: list[MappedPart] = []
    for cat, records in data.items():
        if cat not in cat_ids:
            raise ValueError(f"Категории {cat} нет в БД. Сначала выполните seed")
        parts, skipped = map_records(cat, records, include_unpriced=include_unpriced)
        stats["categories"][cat] = {"read": len(records), "imported": len(parts), "skipped": dict(skipped)}
        mapped.extend(parts)

    makers = {m.manufacturer[0]: m.manufacturer[1] for m in mapped if m.manufacturer}
    for chunk in _chunks(sorted(makers.items()), 1000):
        await session.execute(pg_insert(Manufacturer).values([{"slug": s, "name": n} for s, n in chunk])
                              .on_conflict_do_nothing(index_elements=["slug"]))
    maker_ids = dict((await session.execute(select(Manufacturer.slug, Manufacturer.id))).all())
    existing = {eid: (pid, price) for eid, pid, price in (await session.execute(
        select(Part.external_id, Part.id, Part.price).where(Part.source == SOURCE))).all()}

    def convert(p: Decimal | None) -> Decimal | None:
        if p is None or usd_rate is None:
            return p
        return (p * usd_rate).quantize(Decimal("1"), ROUND_HALF_UP)

    rows = [{
        "category_id": cat_ids[m.category], "manufacturer_id": maker_ids.get(m.manufacturer[0]) if m.manufacturer else None,
        "name": m.name, "source": SOURCE, "external_id": m.external_id, "specs": m.specs,
        "power_w": m.power_w, "price": convert(m.price_usd), "currency": currency,
    } for m in mapped]
    by_eid = {r["external_id"]: r for r in rows}
    table = Part.__table__

    for chunk in _chunks(rows, 500):
        stmt = pg_insert(Part).values(chunk)
        ex = stmt.excluded
        set_ = {
            "category_id": ex.category_id, "manufacturer_id": ex.manufacturer_id, "name": ex.name,
            "specs": ex.specs, "power_w": ex.power_w, "price": ex.price, "currency": ex.currency,
            "is_active": true(),
            "price_updated_at": case((table.c.price.is_distinct_from(ex.price), func.now()),
                                     else_=table.c.price_updated_at),
        }
        if "updated_at" in table.c:
            set_["updated_at"] = func.now()
        stmt = stmt.on_conflict_do_update(index_elements=["source", "external_id"], set_=set_)
        result = (await session.execute(stmt.returning(table.c.id, table.c.external_id))).all()
        history = []
        for pid, eid in result:
            prev, row = existing.get(eid), by_eid[eid]
            stats["created" if prev is None else "updated"] += 1
            if row["price"] is not None and (prev is None or prev[1] != row["price"]):
                history.append({"part_id": pid, "price": row["price"], "currency": currency, "source": SOURCE})
                if prev is not None:
                    stats["price_changes"] += 1
        if history:
            await session.execute(insert(PartPriceHistory), history)

    await session.execute(update(Part).where(Part.source == SOURCE, Part.price.is_not(None),
                                             Part.price_updated_at.is_(None)).values(price_updated_at=func.now()))
    if retire_missing:
        for cat in data:
            seen = [r["external_id"] for r in rows if r["category_id"] == cat_ids[cat]]
            cond = [Part.source == SOURCE, Part.category_id == cat_ids[cat], Part.is_active.is_(True)]
            if seen:
                cond.append(Part.external_id.not_in(seen))
            res = await session.execute(update(Part).where(*cond).values(is_active=False))
            stats["retired"] += res.rowcount or 0
    if retire_seed:
        res = await session.execute(update(Part).where(Part.source == "seed", Part.is_active.is_(True))
                                    .values(is_active=False))
        stats["seed_retired"] = res.rowcount or 0
    return stats

NAMING_CONVENTION = {
    "ix": "ix_%(column_0_label)s",
    "uq": "uq_%(table_name)s_%(column_0_N_name)s",
    "ck": "ck_%(table_name)s_%(constraint_name)s",

    created_at: Mapped[datetime] = mapped_column(server_default=func.now())
    updated_at: Mapped[datetime] = mapped_column(server_default=func.now(), onupdate=func.now())


def str_enum(enum_cls: type[StrEnum], name: str) -> Enum:
    """VARCHAR + CHECK вместо нативного ENUM: новые значения добавляются без ALTER TYPE."""
    return Enum(
        enum_cls,
        name=name,
        native_enum=False,
        create_constraint=True,
        validate_strings=True,
        values_callable=lambda members: [m.value for m in members],
    )

"""Проверка сборки из демо-деталей seed по их ключам.

    docker compose exec api python -m app.scripts.check_build amd-ryzen5-7600 msi-b650-gaming-plus-wifi kingston-fury-ddr5-32-6000:2

Ключ с суффиксом :N задаёт количество. Вывод — одна строка JSON.
"""
import argparse
import asyncio
import json
import sys

from sqlalchemy import select

from app.models import Part
from app.seed_data import SEED_SOURCE
from app.services.build_check import UnknownPartsError, check_items


def _parse(arg: str) -> tuple[str, int]:
    key, _, qty = arg.partition(":")
    return key, int(qty) if qty else 1


async def _main(argv: list[str]) -> int:
    parser = argparse.ArgumentParser(description="Проверка совместимости демо-сборки")
    parser.add_argument("parts", nargs="+", help="ключи деталей из seed, например amd-ryzen5-7600 или kit:2")
    args = parser.parse_args(argv)
    try:
        items = [_parse(a) for a in args.parts]
    except ValueError:
        parser.error("количество после двоеточия должно быть целым числом")

    from app.database import SessionLocal, engine

    try:
        async with SessionLocal() as session:
            keys = [k for k, _ in items]
            ids = dict((await session.execute(
                select(Part.external_id, Part.id).where(Part.source == SEED_SOURCE, Part.external_id.in_(keys))
            )).all())
            unknown = [k for k in keys if k not in ids]
            if unknown:
                print(json.dumps({"error": "unknown_parts", "keys": unknown}, ensure_ascii=False))
                return 2
            try:
                result = await check_items(session, [(ids[k], q) for k, q in items])
            except (UnknownPartsError, ValueError) as exc:
                print(json.dumps({"error": "invalid_items", "message": str(exc)}, ensure_ascii=False))
                return 2
    finally:
        await engine.dispose()
    print(json.dumps(result.to_dict(), ensure_ascii=False))
    return 0


if __name__ == "__main__":
    sys.exit(asyncio.run(_main(sys.argv[1:])))

"""Создание администратора или смена пароля.

    docker compose exec api python -m app.scripts.create_admin --username admin
    docker compose exec api python -m app.scripts.create_admin --username admin --reset-password

Пароль запрашивается без отображения. Для автоматизации можно передать ADMIN_PASSWORD.
"""
import argparse
import asyncio
import getpass
import os
import re
import sys

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.models import Admin
from app.security import hash_password

USERNAME_RE = re.compile(r"^[a-z0-9_.-]{3,64}$")
MIN_PASSWORD_LENGTH = 12


def validate(username: str, password: str) -> None:
    if not USERNAME_RE.fullmatch(username):
        raise ValueError("Логин: 3-64 символа, строчные латинские буквы, цифры, точка, дефис, подчёркивание")
    if len(password) < MIN_PASSWORD_LENGTH:
        raise ValueError(f"Пароль должен быть не короче {MIN_PASSWORD_LENGTH} символов")


async def upsert_admin(session: AsyncSession, username: str, password: str, reset: bool = False) -> str:
    validate(username, password)
    admin = await session.scalar(select(Admin).where(Admin.username == username))
    if admin is None:
        session.add(Admin(username=username, password_hash=hash_password(password)))
        result = "created"
    elif not reset:
        raise ValueError(f"Администратор {username} уже существует. Для смены пароля добавьте --reset-password")
    else:
        admin.password_hash = hash_password(password)
        admin.failed_login_count = 0
        admin.locked_until = None
        admin.is_active = True
        result = "updated"
    await session.commit()
    return result


def _read_password() -> str:
    env_password = os.environ.get("ADMIN_PASSWORD")
    if env_password:
        return env_password
    first = getpass.getpass("Пароль: ")
    if first != getpass.getpass("Повторите пароль: "):
        raise ValueError("Пароли не совпадают")
    return first


async def _main(username: str, reset: bool) -> int:
    from app.database import SessionLocal, engine

    try:
        password = _read_password()
        async with SessionLocal() as session:
            result = await upsert_admin(session, username, password, reset)
    except ValueError as exc:
        print(f"Ошибка: {exc}", file=sys.stderr)
        return 1
    finally:
        await engine.dispose()
    print("Администратор создан" if result == "created" else "Пароль обновлён")
    return 0


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Создать администратора")
    parser.add_argument("--username", required=True)
    parser.add_argument("--reset-password", action="store_true")
    args = parser.parse_args()
    sys.exit(asyncio.run(_main(args.username.strip().lower(), args.reset_password)))

    p.add_argument("--keep-seed", action="store_true", help="не скрывать демо-детали из seed")
    p.add_argument("--dry-run", action="store_true", help="посчитать и откатить транзакцию")
    args = p.parse_args(argv)
    if args.usd_rub_rate is not None:
        try:
            args.usd_rub_rate = Decimal(str(args.usd_rub_rate).replace(",", "."))
        except InvalidOperation:
            p.error("--usd-rub-rate: ожидалось число")
        if args.usd_rub_rate <= 0:
            p.error("--usd-rub-rate должен быть больше нуля")
    elif not args.keep_usd:
        p.error("не задан курс: DOCYX_USD_RUB_RATE в .env или --usd-rub-rate. Для цен в USD добавьте --keep-usd")
    return args

                                             include_unpriced=not args.skip_unpriced,

            else:
                await session.commit()
    finally:
        await engine.dispose()
    stats["dry_run"] = args.dry_run
    print(json.dumps(stats, ensure_ascii=False, indent=2))
    return 0

"""Идемпотентная загрузка стартовых данных.

    docker compose exec api python -m app.scripts.seed

Категории обновляются, правила только добавляются (правки админа сохраняются),
демо-детали обновляются по (source, external_id), смена цены пишется в историю.
"""
import asyncio
import json
from datetime import UTC, datetime

from sqlalchemy import select
from sqlalchemy.dialects.postgresql import insert as pg_insert
from sqlalchemy.ext.asyncio import AsyncSession

from app.models import Category, CompatibilityRule, Manufacturer, Part, PartPriceHistory
from app.seed_data import CATEGORIES, MANUFACTURERS, PARTS, RULES, SEED_SOURCE

CURRENCY = "RUB"


async def seed(session: AsyncSession) -> dict[str, int]:
    stats = {"rules_added": 0, "parts_added": 0, "parts_updated": 0, "prices_recorded": 0}

    for row in CATEGORIES:
        stmt = pg_insert(Category).values(**row)
        await session.execute(
            stmt.on_conflict_do_update(
                index_elements=[Category.slug],
                set_={k: stmt.excluded[k] for k in ("name", "sort_order", "is_required", "max_items")},
            )
        )

    await session.execute(
        pg_insert(Manufacturer).values(MANUFACTURERS).on_conflict_do_nothing(index_elements=[Manufacturer.slug])
    )

    for row in RULES:
        added = await session.scalar(
            pg_insert(CompatibilityRule)
            .values(**row)
            .on_conflict_do_nothing(index_elements=[CompatibilityRule.code])
            .returning(CompatibilityRule.id)
        )
        stats["rules_added"] += added is not None

    category_ids = dict((await session.execute(select(Category.slug, Category.id))).all())
    manufacturer_ids = dict((await session.execute(select(Manufacturer.slug, Manufacturer.id))).all())
    existing = {
        p.external_id: p
        for p in await session.scalars(select(Part).where(Part.source == SEED_SOURCE))
    }
    now = datetime.now(UTC)

    for item in PARTS:
        values = {
            "category_id": category_ids[item["category"]],
            "manufacturer_id": manufacturer_ids[item["manufacturer"]],
            "name": item["name"],
            "power_w": item["power_w"],
            "specs": item["specs"],
        }
        part = existing.get(item["key"])
        if part is None:
            part = Part(source=SEED_SOURCE, external_id=item["key"], currency=CURRENCY,
                        price=item["price"], price_updated_at=now, **values)
            session.add(part)
            await session.flush()
            stats["parts_added"] += 1
            price_changed = True
        else:
            changed = False
            for field, value in values.items():
                if getattr(part, field) != value:
                    setattr(part, field, value)
                    changed = True
            price_changed = part.price != item["price"]
            if price_changed:
                part.price = item["price"]
                part.price_updated_at = now
            stats["parts_updated"] += changed or price_changed

        if price_changed:
            session.add(PartPriceHistory(part_id=part.id, price=item["price"], currency=CURRENCY,
                                         source=SEED_SOURCE))
            stats["prices_recorded"] += 1

    await session.commit()
    return stats


async def _main() -> None:
    from app.database import SessionLocal, engine

    try:
        async with SessionLocal() as session:
            stats = await seed(session)
    finally:
        await engine.dispose()
    print(json.dumps(stats, ensure_ascii=False))


if __name__ == "__main__":
    asyncio.run(_main())

"""Связка движка совместимости с БД: правила, категории и детали из PostgreSQL."""
from collections.abc import Iterable

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from app.models import Category, CompatibilityRule, Part
from app.services.compatibility import CheckResult, RuleConfig, check_build
from app.services.parts import BuildPart

MAX_QUANTITY = 16  # совпадает с build_items.quantity_range
MAX_DISTINCT_PARTS = 64


class UnknownPartsError(LookupError):
    """Деталь не найдена, удалена или снята с продажи."""

    def __init__(self, part_ids: Iterable[int]):
        self.part_ids = tuple(sorted(part_ids))
        super().__init__(f"Детали недоступны: {list(self.part_ids)}")


def normalize_items(items: Iterable[tuple[int, int]]) -> dict[int, int]:
    """Склеивает повторы part_id и проверяет количество. Порядок первого появления сохраняется."""
    result: dict[int, int] = {}
    for part_id, qty in items:
        if isinstance(part_id, bool) or not isinstance(part_id, int) or part_id <= 0:
            raise ValueError(f"Некорректный part_id: {part_id!r}")
        if isinstance(qty, bool) or not isinstance(qty, int) or qty < 1:
            raise ValueError(f"Некорректное количество для {part_id}: {qty!r}")
        result[part_id] = result.get(part_id, 0) + qty
        if result[part_id] > MAX_QUANTITY:
            raise ValueError(f"Количество для {part_id} больше {MAX_QUANTITY}")
    if len(result) > MAX_DISTINCT_PARTS:
        raise ValueError(f"В сборке больше {MAX_DISTINCT_PARTS} разных деталей")
    return result


async def load_rule_configs(session: AsyncSession) -> dict[str, RuleConfig]:
    rows = await session.scalars(select(CompatibilityRule))
    return {r.code: RuleConfig(r.code, r.severity, r.is_enabled, dict(r.params or {})) for r in rows}


async def load_categories(session: AsyncSession) -> tuple[dict[str, int], tuple[str, ...]]:
    rows = (await session.execute(
        select(Category.slug, Category.max_items, Category.is_required).order_by(Category.sort_order)
    )).all()
    return {slug: limit for slug, limit, _ in rows}, tuple(slug for slug, _, req in rows if req)


async def load_build_parts(session: AsyncSession, items: dict[int, int]) -> list[BuildPart]:
    if not items:
        return []
    rows = (await session.execute(
        select(Part, Category.slug)
        .join(Category, Part.category_id == Category.id)
        .where(Part.id.in_(items), Part.deleted_at.is_(None), Part.is_active)
    )).all()
    found = {part.id: (part, slug) for part, slug in rows}
    missing = set(items) - set(found)
    if missing:
        raise UnknownPartsError(missing)
    return [
        BuildPart(part_id=pid, category=found[pid][1], name=found[pid][0].name,
                  specs=found[pid][0].specs or {}, power_w=found[pid][0].power_w,
                  quantity=qty, price=found[pid][0].price)
        for pid, qty in items.items()
    ]


async def check_items(session: AsyncSession, items: Iterable[tuple[int, int]]) -> CheckResult:
    """Проверка набора (part_id, quantity) по правилам и категориям из БД."""
    normalized = normalize_items(items)
    parts = await load_build_parts(session, normalized)
    rules = await load_rule_configs(session)
    limits, required = await load_categories(session)
    return check_build(parts, rules, category_limits=limits, required_categories=required)

"""Движок совместимости. Чистые функции, без БД и сети.

Логика правил в коде, включение, severity и пороги приходят из compatibility_rules.

правило пропускается с предупреждением missing_specs, сборка не блокируется.

MISSING_SPECS = "missing_specs"
DEFAULT_CATEGORY_LIMITS: dict[str, int] = {c["slug"]: c["max_items"] for c in CATEGORIES}

PARAM_LIMITS: dict[str, dict[str, tuple[float, float]]] = {

@dataclass(frozen=True, slots=True)

@dataclass(frozen=True, slots=True)
class Issue:

# ---------- Чтение характеристик ----------

class MissingSpec(Exception):
    def __init__(self, part: BuildPart, key: str):
        super().__init__(f"{part.name}: {key}")
        self.part = part
        self.key = key


def _num(part: BuildPart, key: str) -> float:
    v = part.specs.get(key)
    if isinstance(v, bool) or not isinstance(v, (int, float)) or not math.isfinite(v) or v < 0:
        raise MissingSpec(part, key)
    return v


def _str(part: BuildPart, key: str) -> str:
    v = part.specs.get(key)
    if not isinstance(v, str) or not v.strip():
        raise MissingSpec(part, key)
    return v.strip()


def _list(part: BuildPart, key: str) -> list[str]:
    v = part.specs.get(key)
    if not isinstance(v, list) or not v or not all(isinstance(x, str) and x.strip() for x in v):
        raise MissingSpec(part, key)
    return [x.strip() for x in v]


def _n(value: str) -> str:
    return value.casefold()


def _in(value: str, options: list[str]) -> bool:
    return _n(value) in {_n(o) for o in options}


def _fmt(value: float) -> str:
    return f"{value:g}"


# ---------- Правила ----------

@dataclass(frozen=True, slots=True)
class _Violation:
    hard: bool
    code: str
    message: str
    parts: tuple[BuildPart, ...]
    details: Mapping[str, Any]


def _hard(code: str, message: str, *parts: BuildPart, **details: Any) -> _Violation:
    return _Violation(True, code, message, parts, details)


def _soft(code: str, message: str, *parts: BuildPart, **details: Any) -> _Violation:
    return _Violation(False, code, message, parts, details)


class _Context:
    def __init__(self, parts: tuple[BuildPart, ...], power: PowerEstimate):
        self.parts = parts
        self.power = power
        self.by_category: dict[str, list[BuildPart]] = {}
        for p in parts:
            self.by_category.setdefault(p.category, []).append(p)

    def all(self, category: str) -> list[BuildPart]:
        return self.by_category.get(category, [])

    def one(self, category: str) -> BuildPart | None:
        items = self.by_category.get(category)
        return items[0] if items else None


Params = Mapping[str, Any]
RuleFn = Callable[[_Context, Params], Iterator[_Violation]]

def _cpu_motherboard_socket(ctx: _Context, params: Params) -> Iterator[_Violation]:
    cpu, mb = ctx.one("cpu"), ctx.one("motherboard")
    if not (cpu and mb):
        return
    a, b = _str(cpu, "socket"), _str(mb, "socket")
    if _n(a) != _n(b):
        yield _hard("socket_mismatch",
                    f"Сокет процессора {cpu.name} ({a}) не совпадает с сокетом платы {mb.name} ({b}).",
                    cpu, mb, cpu_socket=a, motherboard_socket=b)


def _ram_motherboard(ctx: _Context, params: Params) -> Iterator[_Violation]:
    rams = ctx.all("ram")
    if not rams:
        return
    cpu, mb = ctx.one("cpu"), ctx.one("motherboard")
    for ram in rams:
        t = _str(ram, "memory_type")
        if mb:
            mb_type = _str(mb, "memory_type")
            if _n(t) != _n(mb_type):
                yield _hard("ram_type_motherboard",
                            f"Плата {mb.name} поддерживает {mb_type}, а память {ram.name} — {t}.",
                            ram, mb, ram_type=t, motherboard_type=mb_type)
        if cpu:
            cpu_types = _list(cpu, "memory_types")
            if not _in(t, cpu_types):
                yield _hard("ram_type_cpu",
                            f"Процессор {cpu.name} не поддерживает {t} (поддерживает {', '.join(cpu_types)}).",
                            ram, cpu, ram_type=t, cpu_types=cpu_types)
    if mb:
        modules = sum(_num(r, "modules") * r.quantity for r in rams)
        slots = _num(mb, "memory_slots")
        if modules > slots:
            yield _hard("ram_slots_exceeded",
                        f"Выбрано {_fmt(modules)} модулей памяти, а на плате {mb.name} {_fmt(slots)} слота.",
                        *rams, mb, modules=modules, slots=slots)
        capacity = sum(_num(r, "capacity_gb") * r.quantity for r in rams)
        max_gb = _num(mb, "max_memory_gb")
        if capacity > max_gb:
            yield _hard("ram_capacity_exceeded",
                        f"Объём памяти {_fmt(capacity)} ГБ больше максимума платы {mb.name} ({_fmt(max_gb)} ГБ).",
                        *rams, mb, capacity_gb=capacity, max_memory_gb=max_gb)
    if len(rams) > 1:
        yield _soft("ram_mixed_kits",
                    "Выбраны разные комплекты памяти. Они могут не заработать на заявленной частоте, "
                    "надёжнее взять один комплект нужного объёма.", *rams)


def _case_form_factors(ctx: _Context, params: Params) -> Iterator[_Violation]:
    case = ctx.one("case")
    if not case:
        return
    mb, psu = ctx.one("motherboard"), ctx.one("psu")
    if mb:
        ff, allowed = _str(mb, "form_factor"), _list(case, "motherboard_form_factors")
        if not _in(ff, allowed):
            yield _hard("case_motherboard_form_factor",
                        f"Корпус {case.name} не поддерживает платы {ff} (поддерживает {', '.join(allowed)}).",
                        mb, case, form_factor=ff, supported=allowed)
    if psu:
        ff, allowed = _str(psu, "form_factor"), _list(case, "psu_form_factors")
        if not _in(ff, allowed):
            if _in(ff, ["SFX", "SFX-L"]) and _in("ATX", allowed):
                yield _soft("case_psu_sfx_adapter",
                            f"{psu.name} формата {ff} встанет в корпус {case.name} под ATX только через "
                            f"переходную рамку SFX→ATX. Проверьте, есть ли она в комплекте БП.",
                            psu, case)
            else:
                yield _hard("case_psu_form_factor",
                            f"Корпус {case.name} рассчитан на БП {', '.join(allowed)}, а {psu.name} — {ff}.",
                            psu, case, form_factor=ff, supported=allowed)


def _clearance(code: str, part: BuildPart, case: BuildPart, size: float, limit: float,
               margin_min: float, what: str) -> Iterator[_Violation]:
    margin = limit - size
    details = {"size_mm": size, "limit_mm": limit, "margin_mm": margin}
    if margin < 0:
        yield _hard(f"{code}_exceeded",
                    f"{what} {part.name} ({_fmt(size)} мм) не помещается в корпус {case.name} (до {_fmt(limit)} мм).",
                    part, case, **details)
    elif margin < margin_min:
        yield _soft(f"{code}_tight",
                    f"{what} {part.name} помещается в корпус {case.name} с запасом всего {_fmt(margin)} мм.",
                    part, case, **details)


def _case_gpu_length(ctx: _Context, params: Params) -> Iterator[_Violation]:
    case, gpus = ctx.one("case"), ctx.all("gpu")
    if not (case and gpus):
        return
    limit = _num(case, "max_gpu_length_mm")
    for gpu in gpus:
        yield from _clearance("gpu_length", gpu, case, _num(gpu, "length_mm"), limit,
                              params["warning_margin_mm"], "Видеокарта")


def _case_cooler_height(ctx: _Context, params: Params) -> Iterator[_Violation]:
    case, cooler = ctx.one("case"), ctx.one("cooler")
    if not (case and cooler):
        return
    yield from _clearance("cooler_height", cooler, case, _num(cooler, "height_mm"),
                          _num(case, "max_cooler_height_mm"), params["warning_margin_mm"], "Кулер")


def _cooler_socket(ctx: _Context, params: Params) -> Iterator[_Violation]:
    cpu, cooler = ctx.one("cpu"), ctx.one("cooler")
    if not (cpu and cooler):
        return
    socket, sockets = _str(cpu, "socket"), _list(cooler, "sockets")
    if not _in(socket, sockets):
        yield _hard("cooler_socket_unsupported",
                    f"Кулер {cooler.name} не поддерживает сокет {socket} (поддерживает {', '.join(sockets)}).",
                    cooler, cpu, socket=socket, supported=sockets)


def _psu_power(ctx: _Context, params: Params) -> Iterator[_Violation]:
    psu = ctx.one("psu")
    if not psu:
        return
    watts, power = _num(psu, "wattage_w"), ctx.power
    details = {"wattage_w": watts, "peak_w": power.peak_w, "recommended_w": power.recommended_w}
    if watts < power.peak_w:
        yield _hard("psu_insufficient",
                    f"Блок питания {psu.name} ({_fmt(watts)} Вт) слабее пиковой нагрузки {power.peak_w} Вт.",
                    psu, **details)
    elif watts < power.recommended_w:
        yield _soft("psu_low_headroom",
                    f"Мощности {psu.name} ({_fmt(watts)} Вт) хватает впритык. Рекомендуется от {power.recommended_w} Вт.",
                    psu, **details)


def _storage_slots(ctx: _Context, params: Params) -> Iterator[_Violation]:
    mb, drives = ctx.one("motherboard"), ctx.all("storage")
    if not (mb and drives):
        return
    counts = {"nvme": 0, "sata": 0}
    for d in drives:
        kind = _n(_str(d, "interface"))
        if kind in counts:
            counts[kind] += d.quantity
    for kind, key, code, label in (("nvme", "m2_slots", "m2_slots_exceeded", "слотов M.2"),
                                   ("sata", "sata_ports", "sata_ports_exceeded", "портов SATA")):
        if not counts[kind]:
            continue
        available = _num(mb, key)
        if counts[kind] > available:
            used = [d for d in drives if _n(_str(d, "interface")) == kind]
            yield _hard(code,
                        f"Накопителей {kind.upper()}: {counts[kind]}, а на плате {mb.name} {_fmt(available)} {label}.",
                        *used, mb, required=counts[kind], available=available)


# Порядок задаёт порядок проверок в ответе
_RULES: dict[str, RuleFn] = {
    "cpu_motherboard_socket": _cpu_motherboard_socket,
    "ram_motherboard": _ram_motherboard,
    "case_form_factors": _case_form_factors,
    "case_gpu_length": _case_gpu_length,
    "case_cooler_height": _case_cooler_height,
    "cooler_socket": _cooler_socket,
    "psu_power": _psu_power,
    "storage_slots": _storage_slots,
}
RULE_CODES: tuple[str, ...] = tuple(_RULES)


# ---------- Конфигурация ----------

def effective_params(code: str, raw: Mapping[str, Any] | None) -> dict[str, Any]:
    defaults = _RULE_DEFAULTS[code]["params"]
    raw = raw or {}
    result: dict[str, Any] = {}
    for key, (lo, hi) in PARAM_LIMITS.get(code, {}).items():
        value = raw.get(key, defaults[key])
        valid = (isinstance(value, (int, float)) and not isinstance(value, bool)
                 and math.isfinite(value) and lo <= value <= hi)
        if not valid:
            logger.warning("Правило %s: некорректный параметр %s=%r, используется %r", code, key, value, defaults[key])
            value = defaults[key]
        result[key] = value
    return result


def resolve_rule(code: str, rules: Mapping[str, RuleConfig] | None) -> RuleConfig:
    cfg = rules.get(code) if rules is not None else None
    if cfg is None:
        if rules is not None:
            logger.warning("Правила %s нет в БД, используются значения по умолчанию", code)
        return RuleConfig(code, Severity(_RULE_DEFAULTS[code]["severity"]), True, effective_params(code, None))
    return RuleConfig(code, Severity(cfg.severity), cfg.is_enabled, effective_params(code, cfg.params))


def _structure_issues(ctx: _Context, limits: Mapping[str, int]) -> Iterator[Issue]:
    for category, items in ctx.by_category.items():
        ids = tuple(p.part_id for p in items)
        qty = sum(p.quantity for p in items)
        limit = limits.get(category)
        if limit is None:
            yield Issue(STRUCTURE_RULE, "unknown_category", Severity.ERROR,
                        f"Неизвестная категория {category}.", ids, {"category": category})
        elif qty > limit:
            yield Issue(STRUCTURE_RULE, "category_limit_exceeded", Severity.ERROR,
                        f"В категории {category} выбрано {qty} шт., допустимо не больше {limit}.",
                        ids, {"category": category, "quantity": qty, "limit": limit})


def check_build(
    parts: Sequence[BuildPart],
    rules: Mapping[str, RuleConfig] | None = None,
    *,
    category_limits: Mapping[str, int] | None = None,
    required_categories: Sequence[str] | None = None,
) -> CheckResult:
    """rules=None — все правила со значениями по умолчанию (для тестов и резервного режима)."""
    parts = tuple(parts)
    limits = DEFAULT_CATEGORY_LIMITS if category_limits is None else category_limits
    required = DEFAULT_REQUIRED_CATEGORIES if required_categories is None else tuple(required_categories)
    configs = {code: resolve_rule(code, rules) for code in _RULES}
    pp = configs["psu_power"].params
    power = estimate_power(parts, base_load_w=pp["base_load_w"], factor=pp["recommended_factor"])
    ctx = _Context(parts, power)

    issues = list(_structure_issues(ctx, limits))
    for code, fn in _RULES.items():
        cfg = configs[code]
        if not cfg.is_enabled:
            continue
        try:
            for v in fn(ctx, cfg.params):
                issues.append(Issue(code, v.code, cfg.severity if v.hard else Severity.WARNING, v.message,
                                    tuple(dict.fromkeys(p.part_id for p in v.parts)), dict(v.details)))
        except MissingSpec as exc:
            issues.append(Issue(
                code, MISSING_SPECS, Severity.WARNING,
                f"Проверка «{_RULE_DEFAULTS[code]['name']}» пропущена: у {exc.part.name} не указано {exc.key}.",
                (exc.part.part_id,), {"key": exc.key},
            ))
    issues.sort(key=lambda i: i.severity != Severity.ERROR)  # сортировка стабильная: ошибки первыми

    present = {p.category for p in parts}
    total = sum((p.price * p.quantity for p in parts if p.price is not None), Decimal("0"))
    return CheckResult(
        issues=tuple(issues),
        power=power,
        missing_categories=tuple(c for c in required if c not in present),
        total_price=total,
        unpriced_part_ids=tuple(p.part_id for p in parts if p.price is None),
    )

"""Оценка потребления сборки и рекомендуемой мощности БП.

peak_w        = base_load_w + сумма(power_w * quantity)
recommended_w = max(ceil(peak_w * factor), рекомендация производителя видеокарты)
suggested_psu_w — ближайшая стандартная мощность БП не меньше recommended_w.
"""
import math
from collections.abc import Iterable, Mapping
from dataclasses import dataclass
from decimal import ROUND_CEILING, Decimal
from typing import Any

from app.services.parts import BuildPart

STANDARD_PSU_W = (300, 400, 450, 500, 550, 600, 650, 700, 750, 850, 1000, 1200, 1300, 1500, 1600)
DEFAULT_BASE_LOAD_W = 50  # плата, вентиляторы, USB
DEFAULT_FACTOR = 1.3


@dataclass(frozen=True, slots=True)
class PowerEstimate:
    peak_w: int
    recommended_w: int
    suggested_psu_w: int
    base_load_w: int
    factor: float
    vendor_min_w: int | None
    by_category: Mapping[str, int]

    def to_dict(self) -> dict[str, Any]:
        return {
            "peak_w": self.peak_w,
            "recommended_w": self.recommended_w,
            "suggested_psu_w": self.suggested_psu_w,
            "base_load_w": self.base_load_w,
            "factor": self.factor,
            "vendor_min_w": self.vendor_min_w,
            "by_category": dict(self.by_category),
        }


def _positive_number(value: Any) -> bool:
    return (
        isinstance(value, (int, float)) and not isinstance(value, bool)
        and math.isfinite(value) and value > 0
    )


def suggest_psu(watts: int) -> int:
    for size in STANDARD_PSU_W:
        if size >= watts:
            return size
    return math.ceil(watts / 100) * 100


def estimate_power(
    parts: Iterable[BuildPart],
    base_load_w: float = DEFAULT_BASE_LOAD_W,
    factor: float = DEFAULT_FACTOR,
) -> PowerEstimate:
    if base_load_w < 0:
        raise ValueError("base_load_w не может быть отрицательным")
    if factor < 1:
        raise ValueError("factor должен быть не меньше 1")
    parts = list(parts)
    by_category: dict[str, int] = {}
    for p in parts:
        by_category[p.category] = by_category.get(p.category, 0) + max(p.power_w or 0, 0) * p.quantity
    base = math.ceil(base_load_w)
    peak = base + sum(by_category.values())
    # Decimal, чтобы 270 * 1.3 давало 351, а не 352 из-за погрешности float
    with_margin = int((Decimal(peak) * Decimal(str(factor))).to_integral_value(rounding=ROUND_CEILING))
    vendor = max(
        (math.ceil(p.specs["recommended_psu_w"]) for p in parts
         if p.category == "gpu" and _positive_number(p.specs.get("recommended_psu_w"))),
        default=None,
    )
    recommended = max(with_margin, vendor or 0)
    return PowerEstimate(peak, recommended, suggest_psu(recommended), base, float(factor), vendor, by_category)

from functools import lru_cache
from typing import Literal

from pydantic import Field, model_validator
from pydantic_settings import BaseSettings, SettingsConfigDict
from sqlalchemy.engine import URL


class Settings(BaseSettings):
    model_config = SettingsConfigDict(env_file=".env", env_file_encoding="utf-8", extra="ignore")

    app_env: Literal["development", "test", "production"] = "development"
    secret_key: str = Field(min_length=32)
    api_prefix: str = "/api/v1"
    cors_origins: str = "http://localhost:8080"

    postgres_user: str
    postgres_password: str
    postgres_db: str
    postgres_host: str = "db"
    postgres_port: int = 5432

    redis_url: str = "redis://redis:6379/0"
    celery_broker_url: str = "redis://redis:6379/1"
    celery_result_backend: str = "redis://redis:6379/2"

    @property
    def database_url(self) -> str:
        # URL.create корректно экранирует спецсимволы в пароле (@ : / #)
        return URL.create(
            drivername="postgresql+asyncpg",
            username=self.postgres_user,
            password=self.postgres_password,
            host=self.postgres_host,
            port=self.postgres_port,
            database=self.postgres_db,
        ).render_as_string(hide_password=False)

    @property
    def cors_origins_list(self) -> list[str]:
        return [o.strip() for o in self.cors_origins.split(",") if o.strip()]

    @property
    def is_production(self) -> bool:
        return self.app_env == "production"

    @model_validator(mode="after")
    def _reject_placeholders_in_production(self) -> "Settings":
        if self.is_production:
            weak = [
                name
                for name, value in (
                    ("SECRET_KEY", self.secret_key),
                    ("POSTGRES_PASSWORD", self.postgres_password),
                )
                if value.startswith("change-me")
            ]
            if weak:
                raise ValueError(f"Placeholder values in production: {', '.join(weak)}")
        return self


@lru_cache
def get_settings() -> Settings:
    return Settings()

===== backend\app\openapi_patch.py =====
"""Дополнение OpenAPI-схемы: Bearer-авторизация для Swagger и адреса серверов.

Подключение в app/main.py сразу после создания app:
    from app.openapi_patch import install_openapi
    install_openapi(app)
Бизнес-логику не меняет: эндпоинты по-прежнему читают заголовок Authorization.
"""
import os
from fastapi import FastAPI
from fastapi.openapi.utils import get_openapi

SCHEME = "AnonToken"


def install_openapi(app: FastAPI) -> None:
    def custom_openapi():
        if app.openapi_schema:
            return app.openapi_schema
        schema = get_openapi(
            title=app.title,
            version=app.version,
            description=app.description,
            routes=app.routes,
        )
        comps = schema.setdefault("components", {})
        comps.setdefault("securitySchemes", {})[SCHEME] = {
            "type": "http",
            "scheme": "bearer",
            "description": "anon_token из ответа POST /api/v1/session",
        }
        for ops in schema.get("paths", {}).values():
            for op in ops.values():
                params = op.get("parameters", [])
                auth = [p for p in params
                        if p.get("in") == "header" and p.get("name", "").lower() == "authorization"]
                if auth:
                    # Заголовок теперь передаётся через кнопку Authorize, а не как поле формы
                    op["parameters"] = [p for p in params if p not in auth]
                    op["security"] = [{SCHEME: []}]
        servers = [u.strip() for u in os.getenv("OPENAPI_SERVERS", "").split(",") if u.strip()]
        if servers:
            schema["servers"] = [{"url": u} for u in servers]
        app.openapi_schema = schema
        return schema

    app.openapi = custom_openapi

"""Alembic: асинхронные миграции через asyncpg."""
import asyncio
from logging.config import fileConfig

from alembic import context
from sqlalchemy.engine import Connection
from sqlalchemy.ext.asyncio import create_async_engine
from sqlalchemy.pool import NullPool

from app.config import get_settings
from app.models import Base

config = context.config
if config.config_file_name and config.attributes.get("configure_logger", True):
    fileConfig(config.config_file_name, disable_existing_loggers=False)

target_metadata = Base.metadata


def _url() -> str:
    # Тесты передают URL отдельной БД через attributes, чтобы не трогать рабочую
    return config.attributes.get("url") or get_settings().database_url


def run_migrations_offline() -> None:
    context.configure(
        url=_url(),
        target_metadata=target_metadata,
        literal_binds=True,
        compare_type=True,
        dialect_opts={"paramstyle": "named"},
    )
    with context.begin_transaction():
        context.run_migrations()


def _run_sync(connection: Connection) -> None:
    context.configure(connection=connection, target_metadata=target_metadata, compare_type=True)
    with context.begin_transaction():
        # Не висеть бесконечно, если таблицу держит долгая транзакция
        connection.exec_driver_sql("SET LOCAL lock_timeout = '10s'")
        context.run_migrations()


async def run_migrations_online() -> None:
    engine = create_async_engine(_url(), poolclass=NullPool)
    try:
        async with engine.connect() as connection:
            await connection.run_sync(_run_sync)
    finally:
        await engine.dispose()


if context.is_offline_mode():
    run_migrations_offline()
else:
    asyncio.run(run_migrations_online())

class DB:
    """Запуск async-кода с отдельной сессией в синхронном тесте."""

    def __init__(self, url: str):
        self.url = url

    def run(self, fn):
        from sqlalchemy.ext.asyncio import AsyncSession, create_async_engine
        from sqlalchemy.pool import NullPool

        async def _go():
            engine = create_async_engine(self.url, poolclass=NullPool)
            try:
                async with AsyncSession(engine, expire_on_commit=False) as session:
                    return await fn(session)
            finally:
                await engine.dispose()

        return asyncio.run(_go())

@pytest.fixture
def seeded(db):
    db.run(seed)
    return db


def run(db, keys, before=None, qty=None):
    """keys: external_id деталей; qty: {external_id: количество}, по умолчанию 1."""
    qty = qty or {}

    async def go(session):
        if before is not None:
            await session.execute(before)
            await session.commit()
        ids = dict((await session.execute(
            select(Part.external_id, Part.id).where(Part.source == SEED_SOURCE, Part.external_id.in_(keys))
        )).all())
        return await check_items(session, [(ids[k], qty.get(k, 1)) for k in keys])
    return db.run(go)


def _rule(code):
    return update(CompatibilityRule).where(CompatibilityRule.code == code)


def codes(result):
    return {i.code for i in result.issues}


def test_seed_base_build_is_complete(seeded):
    result = run(seeded, BASE)
    assert result.is_complete and result.issues == ()
    assert result.power.suggested_psu_w == 550


def test_rule_disabled_in_db(seeded):
    keys = ["intel-i5-14400f", *BASE[1:]]
    assert "socket_mismatch" in codes(run(seeded, keys))
    result = run(seeded, keys, _rule("cpu_motherboard_socket").values(is_enabled=False))
    assert "socket_mismatch" not in codes(result) and result.is_compatible


def test_params_from_db(seeded):
    keys = ["amd-ryzen5-7600", "asrock-b650i-lightning-wifi", "kingston-fury-ddr5-32-6000", "asus-tuf-rtx4080s",
            "samsung-990pro-1tb", "corsair-rm850e", "coolermaster-q300l", "noctua-nh-l9a-am5"]
    assert run(seeded, keys).issues == ()
    result = run(seeded, keys, _rule("case_gpu_length").values(params={"warning_margin_mm": 15}))

def test_severity_from_db(seeded):
    # ASRock B650I: 2 порта SATA и 2 слота M.2. Два SATA-диска помещаются, три уже нет
    keys = [k for k in BASE if k != "msi-b650-gaming-plus-wifi"] + ["asrock-b650i-lightning-wifi", "kingston-nv2-2tb",
                                                                    "samsung-870evo-1tb", "seagate-barracuda-2tb"]
    assert run(seeded, keys).issues == ()

    over = {"samsung-870evo-1tb": 2}  # 2 + 1 = 3 SATA при 2 портах
    result = run(seeded, keys, qty=over)
    assert {i.code for i in result.warnings} == {"sata_ports_exceeded"} and not result.errors

    result = run(seeded, keys, _rule("storage_slots").values(severity=Severity.ERROR), qty=over)
    assert {i.code for i in result.errors} == {"sata_ports_exceeded"} and not result.is_compatible

def P(key, qty=1, **specs):
    pid, item = _BY_KEY[key]
    merged = {k: v for k, v in {**item["specs"], **specs}.items() if v is not None}
    return BuildPart(pid, item["category"], item["name"], merged, item["power_w"], qty, Decimal(str(item["price"])))


def make(**override):
    """make(case="fractal-terra", gpu=None, storage=[("kingston-nv2-2tb", 2)])"""
    slots = {**BASE, **override}
    parts = []
    for value in slots.values():
        if value is None:
            continue
        for entry in value if isinstance(value, list) else [value]:
            parts.append(P(*entry) if isinstance(entry, tuple) else P(entry))
    return parts


def check(parts, **rules):
    cfg = {code: RuleConfig(code, **kw) for code, kw in rules.items()} if rules else None
    return check_build(parts, cfg)


def codes(result, severity=None):
    return {i.code for i in result.issues if severity is None or i.severity == severity}


E, W = Severity.ERROR, Severity.WARNING

def token(client) -> dict[str, str]:
    r = client.post(f"{API}/session")
    assert r.status_code == 201, r.text
    body = r.json()
    assert len(body["anon_token"]) >= 40 and body["builds_limit"] == 15
    return {"Authorization": f"Bearer {body['anon_token']}"}


def items(client, keys):
    return [{"part_id": client.ids[k], "quantity": 1} for k in keys]


def build_body(client, keys=BASE, name="Игровой ПК"):

import type { SessionCreated } from './types';

const BASE = '/api/v1';
const TOKEN_KEY = 'pcc.anon_token';

/** Ошибка API. Бэкенд отвечает {"detail": {"code", "message"}}, а при ошибке валидации {"detail": [...]}. */
export class ApiError extends Error {
  constructor(
    public readonly status: number,
    public readonly code: string,
    message: string,
    public readonly details?: unknown,
  ) {
    super(message);
    this.name = 'ApiError';
  }
}

export type Query = Record<string, string | number | boolean | undefined | null>;

function toQueryString(query?: Query): string {
  if (!query) return '';
  const params = new URLSearchParams();
  for (const [key, value] of Object.entries(query)) {
    if (value !== undefined && value !== null && value !== '') params.set(key, String(value));
  }
  const s = params.toString();
  return s ? `?${s}` : '';
}

async function toApiError(res: Response): Promise<ApiError> {
  let body: unknown = null;
  try {
    body = await res.json();
  } catch {
    // не JSON (например, 502 от nginx)
  }
  const detail = (body as { detail?: unknown } | null)?.detail;
  if (detail && typeof detail === 'object' && !Array.isArray(detail) && 'code' in detail) {
    const d = detail as { code: string; message?: string };
    return new ApiError(res.status, d.code, d.message ?? 'Ошибка запроса', detail);
  }
  if (Array.isArray(detail)) {
    return new ApiError(res.status, 'validation_error', 'Некорректные данные запроса', detail);
  }
  const message = res.status >= 500 ? 'Сервер недоступен, попробуйте позже' : 'Ошибка запроса';
  return new ApiError(res.status, `http_${res.status}`, message);
}

// localStorage может быть недоступен (приватный режим, запрет cookies)
function readToken(): string | null {
  try {
    return localStorage.getItem(TOKEN_KEY);
  } catch {
    return null;
  }
}

function writeToken(token: string | null): void {
  try {
    if (token) localStorage.setItem(TOKEN_KEY, token);
    else localStorage.removeItem(TOKEN_KEY);
  } catch {
    // токен проживёт только до перезагрузки страницы
  }
}

let memoryToken: string | null = null;
let pendingSession: Promise<string> | null = null;

async function createSession(): Promise<string> {
  const res = await fetch(`${BASE}/session`, { method: 'POST', headers: { Accept: 'application/json' } });
  if (!res.ok) throw await toApiError(res);
  const data = (await res.json()) as SessionCreated;
  memoryToken = data.anon_token;
  writeToken(data.anon_token);
  return data.anon_token;
}

/** Возвращает токен гостя, при необходимости создаёт сессию. Параллельные вызовы получают одну сессию. */
export function ensureToken(): Promise<string> {
  const token = memoryToken ?? readToken();
  if (token) return Promise.resolve(token);
  pendingSession ??= createSession().finally(() => {
    pendingSession = null;
  });
  return pendingSession;
}

function resetToken(): void {
  memoryToken = null;
  writeToken(null);
}

export interface RequestOptions {
  method?: 'GET' | 'POST' | 'PUT' | 'DELETE';
  body?: unknown;
  query?: Query;
  auth?: boolean;
  signal?: AbortSignal;
}

export async function request<T>(path: string, opts: RequestOptions = {}): Promise<T> {
  const send = (token?: string) => {
    const headers: Record<string, string> = { Accept: 'application/json' };
    if (opts.body !== undefined) headers['Content-Type'] = 'application/json';
    if (token) headers.Authorization = `Bearer ${token}`;
    return fetch(`${BASE}${path}${toQueryString(opts.query)}`, {
      method: opts.method ?? 'GET',
      headers,
      body: opts.body !== undefined ? JSON.stringify(opts.body) : undefined,
      signal: opts.signal,
    });
  };

  let res = await send(opts.auth ? await ensureToken() : undefined);
  // Токен истёк или удалён из БД: заводим новую сессию и повторяем запрос один раз.
  // Сборки старой сессии при этом недоступны, это ожидаемое поведение гостевого режима.
  if (opts.auth && res.status === 401) {
    resetToken();
    res = await send(await ensureToken());
  }
  if (!res.ok) throw await toApiError(res);
  if (res.status === 204) return undefined as T;
  return (await res.json()) as T;
}

===== frontend\src\components\DebouncedInput.tsx =====
import { useEffect, useState, type InputHTMLAttributes } from 'react';

type Props = Omit<InputHTMLAttributes<HTMLInputElement>, 'value' | 'onChange'> & {
  value: string;
  onChange: (value: string) => void;
  delay?: number;
};

/** Сообщает новое значение с задержкой, чтобы не отправлять запрос на каждую букву. */
export default function DebouncedInput({ value, onChange, delay = 400, ...rest }: Props) {
  const [local, setLocal] = useState(value);

  // значение поменялось снаружи (например, «Сбросить фильтры»)
  useEffect(() => setLocal(value), [value]);

  useEffect(() => {
    if (local === value) return;
    const t = window.setTimeout(() => onChange(local), delay);
    return () => window.clearTimeout(t);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [local]);

  return <input {...rest} value={local} onChange={(e) => setLocal(e.target.value)} />;
}

===== frontend\src\hooks\useAsync.ts =====
import { useCallback, useEffect, useState, type DependencyList } from 'react';
import { ApiError } from '../api/client';

export interface AsyncState<T> {
  data?: T;
  error?: string;
  loading: boolean;
}

export function errorText(e: unknown): string {
  return e instanceof ApiError ? e.message : 'Не удалось связаться с сервером';
}

/**
 * Загружает данные и отменяет запрос, если зависимости изменились или компонент размонтирован.
 * Пока идёт повторная загрузка, старые данные остаются на экране (без мигания при смене страницы).
 */
export function useAsync<T>(
  fn: (signal: AbortSignal) => Promise<T>,
  deps: DependencyList,
): AsyncState<T> & { reload: () => void } {
  const [state, setState] = useState<AsyncState<T>>({ loading: true });
  const [nonce, setNonce] = useState(0);

  useEffect(() => {
    const ctrl = new AbortController();
    setState((s) => ({ data: s.data, loading: true }));
    fn(ctrl.signal).then(
      (data) => {
        if (!ctrl.signal.aborted) setState({ data, loading: false });
      },
      (e: unknown) => {
        if (!ctrl.signal.aborted) setState({ error: errorText(e), loading: false });
      },
    );
    return () => ctrl.abort();
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [...deps, nonce]);

  const reload = useCallback(() => setNonce((n) => n + 1), []);
  return { ...state, reload };
}

export type PluralForms = [one: string, few: string, many: string];

export function plural(n: number, forms: PluralForms): string {
  const a = Math.abs(n) % 100;
  const b = a % 10;
  if (a > 10 && a < 20) return forms[2];
  if (b > 1 && b < 5) return forms[1];
  if (b === 1) return forms[0];
  return forms[2];
}

export const countText = (n: number, forms: PluralForms): string =>
  `${n.toLocaleString('ru-RU')} ${plural(n, forms)}`;

export const PARTS_FORMS: PluralForms = ['товар', 'товара', 'товаров'];

export function specValue(key: string, value: string): string {
  if (key === 'interface') return value === 'nvme' ? 'NVMe' : value === 'sata' ? 'SATA' : value;
  if (key === 'form_factor' && /^\d+(\.\d+)?$/.test(value)) return `${value}″`;
  return value;
}

  { value: '-price', label: 'Сначала дорогие' },
  { value: 'name', label: 'По названию' },

  { key: 'form_factor', label: 'Форм-фактор' },
  { key: 'memory_type', label: 'Тип памяти' },

type FacetKey = (typeof FACETS)[number]['key'];

const FILTER_KEYS = ['q', 'manufacturer', 'min_price', 'max_price', 'has_price', 'socket', 'form_factor', 'memory_type'];

function toPrice(v: string | null): number | undefined {
  if (!v) return undefined;
  const n = Number(v);
  return Number.isFinite(n) && n >= 0 ? n : undefined;
}

// Если поле в specs называется не так, как параметр запроса (массивы).
// Бэкенд ищет по ним (проверено: cpu+DDR4 → 3, cooler+LGA1700 → 2, case+ATX → 2).
const SPEC_SOURCE: Record<string, Partial<Record<FacetKey, { spec: string; label: string }>>> = {
  cpu: { memory_type: { spec: 'memory_types', label: 'Тип памяти' } },
  cooler: { socket: { spec: 'sockets', label: 'Сокет' } },
  case: { form_factor: { spec: 'motherboard_form_factors', label: 'Форм-фактор платы' } },
};

function facetOptions(parts: Part[], slug: string, key: FacetKey) {
  const src = SPEC_SOURCE[slug]?.[key];
  const specKey = src?.spec ?? key;
  const values = new Set<string>();
  let label = src?.label;
  for (const p of parts) {
    const v = p.specs[specKey];
    if (typeof v === 'string' && v) values.add(v);
    else if (Array.isArray(v)) for (const x of v) if (typeof x === 'string' && x) values.add(x);
    label ??= p.specs_display.find((s) => s.key === specKey)?.label;
  }
  return { label, values: [...values].sort((a, b) => a.localeCompare(b, 'ru', { numeric: true })) };
}

    limit: PAGE_SIZE,
    offset: (page - 1) * PAGE_SIZE,

===== frontend\vite.config.ts =====
import { defineConfig } from 'vite';
import react from '@vitejs/plugin-react';

// В разработке запросы /api уходят в nginx из docker compose.
// Браузер видит один origin, поэтому CORS не нужен.
export default defineConfig({
  plugins: [react()],
  server: {
    port: 5173,
    strictPort: true,
    proxy: {
      '/api': { target: 'http://127.0.0.1:8080', changeOrigin: true },
    },
  },
});

===== nginx\default.conf =====
server {
    listen 80;
    server_name _;
    server_tokens off;
    client_max_body_size 2m;

    # Встроенный DNS Docker: после пересборки api nginx найдёт новый IP контейнера
    resolver 127.0.0.11 valid=10s ipv6=off;
    set $api_backend http://api:8000;

    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "DENY" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;

    gzip on;
    gzip_types application/json text/plain text/css application/javascript;

    location /api/ {
        proxy_pass $api_backend;
        proxy_http_version 1.1;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $remote_addr;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_connect_timeout 5s;
        proxy_read_timeout 60s;
    }

    # Заглушка до этапа с фронтендом
    location / {
        default_type text/plain;
        return 200 "pc-configurator: frontend is not deployed yet\n";
    }
}
